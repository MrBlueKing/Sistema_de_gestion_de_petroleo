1 — Stock actual de todos los estanques por faena

Foto instantánea del nivel de cada estanque: capacidad, stock actual, porcentaje de llenado y última medición.

SELECT
  f.id_faena,
  f.ubicacion AS faena,
  e.id_estanque,
  e.nombre AS estanque,
  e.tipo_estanque,
  e.capacidad,
  es.stock_actual,
  es.altura_actual,
  ROUND((es.stock_actual / e.capacidad) * 100, 1) AS porcentaje_lleno,
  es.ultima_medicion,
  CASE
    WHEN (es.stock_actual / e.capacidad) >= 0.75 THEN 'OK'
    WHEN (es.stock_actual / e.capacidad) >= 0.25 THEN 'MEDIO'
    ELSE 'BAJO'
  END AS nivel_alerta
FROM estanques e
JOIN faenas f ON f.id_faena = e.id_faena
JOIN estanque_stock es ON es.id_estanque = e.id_estanque
WHERE e.activo = 1
-- AND f.id_faena = :id_faena ← descomentar para filtrar por faena
ORDER BY f.ubicacion, e.tipo_estanque DESC, e.nombre;

2 — Ingresos del período (filtrable por fecha y faena)

Todos los ingresos con su detalle: guía, litros documentados vs recibidos, diferencia, precio y a qué estanque llegó cada uno.

SELECT
  f.ubicacion AS faena,
  ip.id_ingreso,
  ip.created_at AS fecha_ingreso,
  ip.numero_guia,
  ip.litros_documentados,
  ip.litros_recibidos,
  ip.diferencia AS diferencia_litros,
  ip.precio,
  ip.precio_total,
  ip.tipo_recepcion,
  -- Estanques que recibieron (via verificacion_altura)
  GROUP_CONCAT(
    DISTINCT CONCAT(e.nombre, ' (', va.volumen_calculado, ' L)')
    ORDER BY va.id_verificacion
    SEPARATOR ' | '
  ) AS distribucion_estanques
FROM ingresos_petroleo ip
JOIN faenas f ON f.id_faena = ip.id_faena
LEFT JOIN verificacion_altura va ON va.id_ingreso = ip.id_ingreso
                             AND va.volumen_calculado > 0
LEFT JOIN estanques e ON e.id_estanque = va.id_estanque
WHERE ip.created_at BETWEEN ':fecha_inicio' AND ':fecha_fin'
-- AND ip.id_faena = :id_faena ← descomentar para filtrar por faena
GROUP BY ip.id_ingreso
ORDER BY ip.created_at DESC;

Todos los trasvasijes del período: qué fijo aportó, cuánto, hacia dónde, stock antes y después.

-- Trasvasijes fijo → móvil (con detalle por estanque fijo)
SELECT

  t.observaciones

JOIN trasvasijes_multi_fijo_a_movil_detalle d ON d.id_trasvasije_multi = t.id_trasvasije
JOIN estanques e_movil ON e_movil.id_estanque = t.id_estanque_movil
JOIN estanques e_fijo ON e_fijo.id_estanque = d.id_estanque_fijo
JOIN faenas f ON f.id_faena = t.id_faena
WHERE t.fecha_trasvasije BETWEEN ':fecha_inicio' AND ':fecha_fin'
-- AND t.id_faena = :id_faena

UNION ALL

4 — Trazabilidad completa: ingreso → estanque → qué pasó después

Por cada ingreso, muestra todos los eventos posteriores en ese estanque: consumos y trasvasijes, en orden cronológico.

WITH ingresos_base AS (
  SELECT
    ip.id_ingreso, ip.numero_guia, ip.created_at AS fecha_ingreso,
    ip.litros_recibidos, ip.precio_total, ip.id_faena,
    va.id_estanque, va.volumen_calculado AS litros_a_este_estanque
  FROM ingresos_petroleo ip
  JOIN verificacion_altura va ON va.id_ingreso = ip.id_ingreso
                           AND va.volumen_calculado > 0
  WHERE ip.created_at BETWEEN ':fecha_inicio' AND ':fecha_fin'
-- AND ip.id_faena = :id_faena
),
eventos AS (
  -- Consumos simples
  SELECT c.id_estanque, c.fecha_consumo AS fecha_evento,
    'CONSUMO' AS tipo,
    c.litros_consumidos AS litros,
    CONCAT('Equipo: ', IFNULL(m.nombre_maquina, 'N/A')) AS detalle
  FROM consumos c
  LEFT JOIN maquinas m ON m.id_maquina = c.id_equipo
  WHERE c.tipo_consumo = 'simple'
  UNION ALL
  -- Consumos múltiples
  SELECT ce.id_estanque, c.fecha_consumo,
    'CONSUMO MÚLTIPLE',
    ce.litros_desde_estanque,
    CONCAT('Equipo: ', IFNULL(m.nombre_maquina, 'N/A'))
  FROM consumo_estanques ce
  JOIN consumos c ON c.id_consumo = ce.id_consumo
  LEFT JOIN maquinas m ON m.id_maquina = c.id_equipo
  UNION ALL
  -- Trasvasijes salida
  SELECT d.id_estanque_fijo, t.fecha_trasvasije,
    'TRASVASIJE SALIDA',
    d.litros_desde_estanque,
    CONCAT('→ ', e_dest.nombre)
  FROM trasvasijes_multi_fijo_a_movil_detalle d
  JOIN trasvasijes_multi_fijo_a_movil t ON t.id_trasvasije = d.id_trasvasije_multi
  JOIN estanques e_dest ON e_dest.id_estanque = t.id_estanque_movil
)
SELECT
  ib.numero_guia,
  DATE(ib.fecha_ingreso) AS fecha_ingreso,
  ib.litros_a_este_estanque AS litros_ingresados,
  e.nombre AS estanque,
  e.tipo_estanque,
  ev.tipo AS tipo_evento,
  DATE(ev.fecha_evento) AS fecha_evento,
  ev.litros AS litros_evento,
  ev.detalle
FROM ingresos_base ib
JOIN estanques e ON e.id_estanque = ib.id_estanque
LEFT JOIN eventos ev ON ev.id_estanque = ib.id_estanque
                 AND ev.fecha_evento >= ib.fecha_ingreso
ORDER BY ib.fecha_ingreso, ib.id_ingreso, ev.fecha_evento;

5 — Consumos por equipo / máquina del período

Ranking de consumo por máquina: litros totales, costo, horas/km registrados y promedio de consumo.

SELECT
  f.ubicacion AS faena,
  m.id_maquina,
  m.nombre_maquina AS equipo,
  m.patente,
  cat.nombre_categoria AS categoria,
  COUNT(c.id_consumo) AS n_cargas,
  SUM(c.litros_consumidos) AS total_litros,
  SUM(c.litros_consumidos * c.precio_litro) AS costo_total,
  MIN(c.fecha_consumo) AS primera_carga,
  MAX(c.fecha_consumo) AS ultima_carga,
  MAX(c.numeral_final) - MIN(c.numeral_inicial) AS diferencia_numerales,
  ROUND(
    SUM(c.litros_consumidos) /
    NULLIF(MAX(c.numeral_final) - MIN(c.numeral_inicial), 0),
    3
  ) AS rendimiento_lts_por_numeral
FROM consumos c
JOIN faenas f ON f.id_faena = c.id_faena
JOIN maquinas m ON m.id_maquina = c.id_equipo
LEFT JOIN categorias_maquinas cat ON cat.id_categoria = m.id_categoria
WHERE c.fecha_consumo BETWEEN ':fecha_inicio' AND ':fecha_fin'
-- AND c.id_faena = :id_faena
GROUP BY m.id_maquina, f.id_faena
ORDER BY total_litros DESC;

6 — Reconciliación: stock sistema vs stock calculado por movimientos

Detecta discrepancias entre lo que dice el sistema y lo que debería haber según ingresos − consumos − trasvasijes. Diferencia positiva = sistema tiene más de lo esperado. Negativa = sistema tiene menos.

SELECT
  f.ubicacion AS faena,
  e.id_estanque,
  e.nombre AS estanque,
  e.tipo_estanque,
  e.capacidad,
  es.stock_actual AS stock_sistema,
  COALESCE(ing.total_ingresado, 0) AS total_ingresado,
  COALESCE(con.consumido_simple, 0) AS consumido_simple,
  COALESCE(conm.consumido_multiple, 0) AS consumido_multiple,
  COALESCE(tras_sal.salida_trasvasije, 0) AS salida_trasvasije,
  COALESCE(tras_ent.entrada_trasvasije, 0) AS entrada_trasvasije,
  ROUND(
    COALESCE(ing.total_ingresado, 0)
    - COALESCE(con.consumido_simple, 0)
    - COALESCE(conm.consumido_multiple, 0)
    - COALESCE(tras_sal.salida_trasvasije, 0)
    + COALESCE(tras_ent.entrada_trasvasije, 0),
    2
  ) AS stock_calculado,
  ROUND(
    es.stock_actual - (
      COALESCE(ing.total_ingresado, 0)
      - COALESCE(con.consumido_simple, 0)
      - COALESCE(conm.consumido_multiple, 0)
      - COALESCE(tras_sal.salida_trasvasije, 0)
      + COALESCE(tras_ent.entrada_trasvasije, 0)
    ), 2
  ) AS diferencia,
  CASE
    WHEN ABS(es.stock_actual - COALESCE(ing.total_ingresado,0) + COALESCE(con.consumido_simple,0) + COALESCE(conm.consumido_multiple,0) + COALESCE(tras_sal.salida_trasvasije,0) - COALESCE(tras_ent.entrada_trasvasije,0)) <= 5 THEN 'OK'
    WHEN ABS(es.stock_actual - COALESCE(ing.total_ingresado,0) + COALESCE(con.consumido_simple,0) + COALESCE(conm.consumido_multiple,0) + COALESCE(tras_sal.salida_trasvasije,0) - COALESCE(tras_ent.entrada_trasvasije,0)) <= 50 THEN 'REVISAR'
    ELSE 'ALERTA'
  END AS estado
FROM estanques e
JOIN faenas f ON f.id_faena = e.id_faena
JOIN estanque_stock es ON es.id_estanque = e.id_estanque
LEFT JOIN (
  SELECT va.id_estanque, SUM(va.volumen_calculado) AS total_ingresado
  FROM verificacion_altura va
  JOIN ingresos_petroleo ip ON ip.id_ingreso = va.id_ingreso
  WHERE va.volumen_calculado > 0
  GROUP BY va.id_estanque
) ing ON ing.id_estanque = e.id_estanque
LEFT JOIN (
  SELECT id_estanque, SUM(litros_consumidos) AS consumido_simple
  FROM consumos WHERE tipo_consumo = 'simple' GROUP BY id_estanque
) con ON con.id_estanque = e.id_estanque
LEFT JOIN (
  SELECT id_estanque, SUM(litros_desde_estanque) AS consumido_multiple
  FROM consumo_estanques GROUP BY id_estanque
) conm ON conm.id_estanque = e.id_estanque
LEFT JOIN (
  SELECT d.id_estanque_fijo, SUM(d.litros_desde_estanque) AS salida_trasvasije
  FROM trasvasijes_multi_fijo_a_movil_detalle d GROUP BY d.id_estanque_fijo
) tras_sal ON tras_sal.id_estanque_fijo = e.id_estanque
LEFT JOIN (
  SELECT id_estanque_fijo, SUM(litros_trasvasijados) AS entrada_trasvasije
  FROM trasvasijes_movil_a_fijo GROUP BY id_estanque_fijo
) tras_ent ON tras_ent.id_estanque_fijo = e.id_estanque
WHERE e.activo = 1
-- AND f.id_faena = :id_faena
ORDER BY estado DESC, f.ubicacion, e.nombre;

7 — Resumen gerencia: costos y totales del período

Vista ejecutiva: total comprado, total consumido, costo, stock valorizado actual y litros sin registrar (tickets pendientes).

SELECT
  f.id_faena,
  f.ubicacion AS faena,
  -- Ingresos del período
  COUNT(DISTINCT ip.id_ingreso) AS n_ingresos,
  SUM(ip.litros_recibidos) AS total_litros_comprados,
  SUM(ip.precio_total) AS gasto_total_compra,
  ROUND(SUM(ip.precio_total) / SUM(ip.litros_recibidos), 2) AS precio_promedio_litro,
  -- Diferencias de recepción (documentado vs recibido)
  SUM(ip.diferencia) AS diferencia_total_recepcion,
  -- Consumos del período
  COUNT(DISTINCT c.id_consumo) AS n_consumos,
  SUM(c.litros_consumidos) AS total_litros_consumidos,
  SUM(c.litros_consumidos * c.precio_litro) AS costo_total_consumo,
  -- Stock actual valorizado
  SUM(es.stock_actual) AS stock_total_litros,
  ROUND(
    SUM(es.stock_actual) * (SUM(ip.precio_total) / NULLIF(SUM(ip.litros_recibidos),0)),
    0
  ) AS stock_valorizado
FROM faenas f
LEFT JOIN ingresos_petroleo ip ON ip.id_faena = f.id_faena
  AND ip.created_at BETWEEN ':fecha_inicio' AND ':fecha_fin'
LEFT JOIN consumos c ON c.id_faena = f.id_faena
  AND c.fecha_consumo BETWEEN ':fecha_inicio' AND ':fecha_fin'
LEFT JOIN estanques e ON e.id_faena = f.id_faena AND e.activo = 1
LEFT JOIN estanque_stock es ON es.id_estanque = e.id_estanque
-- AND f.id_faena = :id_faena
GROUP BY f.id_faena
ORDER BY gasto_total_compra DESC;
